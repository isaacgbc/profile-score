export function normalizeLinkedinUrl(raw: string): string {
  const url = new URL(raw.trim());
  const host = url.hostname.toLowerCase().replace(/^(m|www|[a-z]{2})\./, "");
  if (host !== "linkedin.com") throw new Error("APIFY_INVALID_URL");
  let pathname = url.pathname.toLowerCase().replace(/\/+$/, "");
  if (!pathname.startsWith("/in/")) throw new Error("APIFY_INVALID_URL");
  const m = pathname.match(/^\/in\/([^/]+)/);
  if (!m) throw new Error("APIFY_INVALID_URL");
  return `https://linkedin.com/in/${m[1]}`;
}
```

       import crypto from "crypto";
       import { prisma } from "@/lib/db/client";

       export const APIFY_CACHE_TTL_MS = 24 * 60 * 60 * 1000; // 24 hours

       export function normalizeLinkedinUrl(raw: string): string { /* from research example */ }
       export function hashUrl(url: string): string { return crypto.createHash("sha256").update(url).digest("hex"); }

       export async function readCache(normalizedUrl: string): Promise<unknown | null> {
         const row = await prisma.linkedInProfileCache.findUnique({ where: { url: normalizedUrl } });
         if (!row) return null;
         if (row.expiresAt.getTime() < Date.now()) return null; // TTL expired → treat as miss
         return row.profileData;
       }

       export async function writeCache(params: {
         normalizedUrl: string;
         profileData: unknown;
         apifyRunId: string | null;
         costUsd: number;
       }): Promise<void> {
         const urlHash = hashUrl(params.normalizedUrl);
         const now = new Date();
         const expiresAt = new Date(now.getTime() + APIFY_CACHE_TTL_MS);
         await prisma.linkedInProfileCache.upsert({
           where: { url: params.normalizedUrl },
           create: {
             url: params.normalizedUrl,
             urlHash,
             profileData: params.profileData as any,
             scrapedAt: now,
             expiresAt,
             apifyRunId: params.apifyRunId,
             costUsd: params.costUsd as any,
           },
           update: {
             urlHash,
             profileData: params.profileData as any,
             scrapedAt: now,
             expiresAt,
             apifyRunId: params.apifyRunId,
             costUsd: params.costUsd as any,
           },
         });
       }
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User URL → Normalizer | Raw LinkedIn URL enters `normalizeLinkedinUrl`; returns canonical form OR throws |
| Normalizer → Cache | Canonical URL is used as DB primary key for cache lookup |
| IP+UA → Fingerprint | Request metadata enters `computeFingerprint`; never logged raw |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 (T-09: SSRF via crafted URL) | Information Disclosure | `normalizeLinkedinUrl` | mitigate | Strict host check `host !== "linkedin.com"` after stripping `m./www./[a-z]{2}` prefixes. Rejects non-linkedin hosts. Rejects non-/in/ paths. Test 7-9 enforce. |
| T-03-02 (T-10: cache poisoning via URL aliasing) | Tampering | `normalizeLinkedinUrl` + `@unique` constraint | mitigate | Deterministic normalization produces ONE canonical form; DB `@unique` on `url` field prevents duplicate rows. Both layers tested (tests 1-6). |
| T-03-03 (T-02: PII over-retention) | Information Disclosure | `readCache` TTL check | mitigate | `readCache` returns `null` when `expiresAt < now` even if row exists in DB. This is the application-layer enforcement; DB cleanup sweep is deferred to a future phase. Tested via Test 13. |
| T-03-04 (T-11 partial: anonymous quota bypass via fingerprint rotation) | Elevation of Privilege | `computeFingerprint` | accept | Residual risk. Salt + UA mixing reduces collisions but a motivated attacker can rotate. Accepted per CONTEXT.md §Deferred Ideas. Rate limiter in Plan 01-07 provides second layer. |
| T-03-05 (T-07: log injection) | Tampering | `apify-cache.ts` errors | mitigate | Errors throw `new Error("APIFY_INVALID_URL")` — a static string code, no user input interpolation. Safe for `logError()` consumption. |
| T-03-06 (T-03: salt exposure) | Information Disclosure | `APIFY_FINGERPRINT_SALT` | mitigate | Read from `process.env` server-only. Never prefixed `NEXT_PUBLIC_`. Default fallback `"ps-apify-v1"` is documented as dev-only. |
